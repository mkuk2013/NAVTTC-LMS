            const html = htmlModel.getValue();

            const js = jsModel.getValue();

            let combined = '<!DOCTYPE html><html><head><style>' + css + '</style></head>';
            combined += '<body>' + html + jsContent + '<' + '/body><' + '/html>';

            if (!iframe) return;

            doc.open();
            doc.write(combined);
            doc.close();
